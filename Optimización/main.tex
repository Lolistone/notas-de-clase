	\section{Introducción} 
	
	Dada $f : \Omega \longrightarrow \mathbb{R}$ nuestro objetivo será encontrar un $x^{*} \in \Omega$ tal que $f(x^{*}) \leq f(x)$ para todo $x \in \Omega$. Dependiendo de nuestro conocimiento acerca de $f$ y $\Omega$ podremos encarar el problema desde distintas perspectivas. Un ejemplo relevante es la optimización convexa ($\Omega$ y $f$ convexos), la cual es suficientemente general para abarcar problemas reales y a su vez, nos permite caracterizar su mínimos y dar algoritmos para hallarlos. 
	
	\subsection{Condiciones de optimalidad}
	
	La idea es comenzar estudiando el caso en que no tenemos restricciones sobre $\Omega$ e ir dando condiciones necesarias y suficientes para ser mínimo. Comencemos viendo que significa ser una solución a un problema de \emph{optimización}.
	
	\begin{definicion}
		Dado un conjunto $\Omega$ y $f : \Omega \longrightarrow \mathbb{R}$ decimos que $x^{*} \in \Omega$ es un minimizador de $f$ si $f(x^{*}) \leq f(x)$ para todo $x \in \Omega$.
	\end{definicion}
	
	
	\begin{definicion}
		Dado un conjunto $\Omega$ y $f : \Omega \longrightarrow \mathbb{R}$ decimos que $x^{*} \in \Omega$ es un minimizador local de $f$ si existe un entorno $\mathcal{N} \subset \Omega$ de $x^{*}$ tal que $f(x^{*}) \leq f(x)$ para todo $x \in \mathcal{N}$ con $x \neq x^{*}$.
	\end{definicion}
	
	Para obtener las condiciones necesarias de optimalidad una idea básica es pensar en el comportamiento de la función en alguna dirección dada. Sea $d \in \mathbb{R}^{n}$ una dirección decimos que es factible si existe $\lambda \in \mathbb{R}$ tal que $x + td \in \Omega$ para todo $0 \leq t \leq \lambda$. Esto nos permite pensar a nuestra función como una función de una variable en el siguiente sentido:
	\[
	\begin{aligned}
		g : [0, \lambda] \to \mathbb{R}, \\ 
		t \mapsto f(x + td).		
	\end{aligned}
	\]
	¿Para qué nos puede interesar esto? Para utilizar cálculo en una variable. 
	
	\begin{prop}[Condiciones necesarias de primer orden] 
		Sea $f : \Omega \subset \mathbb{R}^{n} \to \mathbb{R}$ con $f \in C^{1}(\Omega)$. Si $x^{*}$ es un mínimizador local de $f$ en $\Omega$, entonces para cualquier dirección factible $d \in \mathbb{R}^{n}$ tenemos que 
		$\nabla f(x^{*}) d \geq 0.$
	\end{prop}
	
	\begin{proof}
		Definiendo $g$ como antes tenemos que tiene un mínimo en $t=0$. Además, sabemos que
		$$
		g(t) - g(0) = g'(0)t + o(t)
		$$
		donde $o(t)$ es un término que va a cero más rápido que $t$. Luego, cómo 0 es mínimo para $t \in [0, \lambda]$
		$$
		0 \leq \frac{g(t) - g(0)}{t} = g'(0) + \frac{o(t)}{t}
		$$
		tomando límite con $t \to 0^{+}$ tenemos el resultado.
	\end{proof}
	
	Un caso importante es en el que $x^{*} \in \Omega$ es un punto interior, cómo cuando $\Omega = \mathbb{R}^{n}$. En este, toda dirección es factible. En consecuencia, tenemos que $\nabla f(x^{*}) = 0$ pues dado $d \in \mathbb{R}^{n}$, $d$ y $-d$ son factibles y por lo que acabamos de ver $\nabla f(x^{*}) d \geq 0$ y $- \nabla f(x^{*}) d \geq 0$. 
	
	\begin{prop}[Condiciones necesarias de segundo orden]
		Sea $\Omega \subset \mathbb{R}^{n}$ y $f \in C^{2}(\Omega)$. Si $x^{*}$ es un minimizador local, entonces para cualquier dirección factible $d$ tenemos que: 
 		\begin{enumerate}
 			\item $\nabla f(x^{*}) d \geq 0$ 
 			\item si $\nabla f(x^{*})d = 0$, entonces $d^{T} \nabla^{2}f(x^{*}) d \geq 0$
 		\end{enumerate}
	\end{prop}
	
	\begin{proof}
		La primer condición ya la probamos, concentremonos en 2. Notemos $\gamma(t) = x + td$ y, $g(t) = f(\gamma(t))$. Tenemos así que
		$$
		g(t) - g(0) = \frac{1}{2}g''(0) + o(t^{2})
		$$
		pues $g'(0) = 0$. Luego, diviendo por $t^{2}$ y escribiendo quién es $g''$ obtenemos para $t$ suficientemente chico
		$$
		0 \leq \frac{g(t) - g(0)}{t^2} = d^{T} \nabla^{2}f(x^{*})d + \frac{o(t^2)}{t^2}
		$$
		tomando límite en $t \to 0$ tenemos el resultado.
	\end{proof}
	
	Vamos ahora a dar algunas condiciones \emph{suficientes} de optimalidad. Primero para el caso general, es decir, el de optimización sin restricciones.
	
	\begin{prop}[Condiciones suficientes de segundo orden - sin restricciones]
		Sea $f \in C^{2}$ uan función definida en una región donde $x^{*}$ es un punto interior. Supongamos además que
		\begin{enumerate}
			\item $\nabla f(x^{*}) = 0$
			\item $\nabla^{2}f(x^{*})$ es definida positiva.
		\end{enumerate}
		Luego, \, $x^{*}$ es un mínimo local estricto de $f$.
 	\end{prop}
 	
 	\begin{proof}
 		La prueba consiste en recordar que si una matriz es definida postiva entonces existe una constante positiva $a \in \mathbb{R}$ tal que para todo $d \in \mathbb{R}^{n}$ vale que $d^{T} \nabla^{2} f(x^{*})d \geq a \|d\| ^{2}$. Luego, por Taylor, 
 		$$
 		f(x^{*} + d) - f(x^{*}) = \frac{1}{2} d^{T} \nabla^{2} f(x^{*}) d \|d\| ^{2} + o(\|d\|^{2}) \geq \frac{a}{2} \| d\|^{2} + o(\|d\|^2)
 		$$
 		donde para $\| d\|$ suficientemente chico el primer termino de la derecha dormindo al segundo y por lo tanto es postivo.
	\end{proof} 	
	
 	\subsection{Convexidad y funciones convexas}
 	
 	Los problemas de optimización convexa son los más generales, desde algún punto de vista, sobre los cuales tenemos gran conocimiento acerca de sus soluciones. Ahora, ¿Qué es la \emph{convexidad}?
 	
 	\begin{definicion}
 		Decimos que un conjunto $\Omega$ es convexo si dados $x, y \in \Omega$ y $\lambda \in [0, 1]$ tenemos que $\lambda x + (1 - \lambda)y \in \Omega$.
 	\end{definicion}	
 	
 	\begin{prop} 
 		Algunas propiedades básicas de los conjuntos convexos.
		\begin{enumerate}
			\item Dado $K \subset \mathbb{R}^n$ convexo y $a \geq 0$, $aK$ es convexo.
			\item Dados $K_1, K_2 \subset \mathbb{R}^n$ conjuntos convexos, entonces $K_1 + K_2$ es convexo.
			\item Dados $K \subset \mathbb{R}^n$ convexo, $k + v$ es convexo para todo $v \in \mathbb{R}^{n}$. 
			\item Dada una familia de subconjuntos convexos de $\mathbb{R}^n $, $\{ K_\alpha \}_\alpha$ se tiene que la intersección $\bigcap_{\alpha} K_\alpha$ es convexa.
		\end{enumerate} 		
 	\end{prop}
 	
 	\begin{proof}
 		Trivial.
 	\end{proof}
 		
 	\begin{definicion}
 		Dado un conjunto $U \subset \mathbb{R}^{n}$ llamamos capsula convexa al conjunto
 		$$
 		\{\sum_{i}^{m} \lambda_i u_i : \sum_{i}^{m} \lambda_i = 1, \lambda_i \geq 0, u_i \in U \}
 		$$
 		y lo notamos conv($U$).
 	\end{definicion}	
 	
 	\begin{obs}
 		Dado $U \subset \mathbb{R}^{n}$ su cápsula convexa es el menor conjunto convexo que lo contiene. 
 	\end{obs}
 	
	\begin{definicion}
		Una función $f$ definida sobre un conjunto convexo $\Omega$ se dice convexa si para todo $x, y \in \Omega$ y, para todo $\lambda$ entre $0$ y $1$ vale que
		$$
		f(\lambda x + (1- \lambda) y) \leq \lambda f(x) + (1-\lambda) f(y)
		$$
		decimos que $f$ es convexa. Además, si para todo $\lambda \in (0, 1)$, $x \neq y$ vale la desigualdad estricta decimos que $f$ es estrictamente convexa.
	\end{definicion}
	
	\begin{obs}
		Nótese que solo tiene sentido hablas de funciones convexas si están definidas sobre conjuntos convexos.
	\end{obs}
	
	Geométricamente podemos pensar que una función es convexa si dados dos puntos en su gráfico la linea que los une está por encima de este. Vale la pena mencionar que una función $f$ es cóncava si $-f$ es convexa.
	
	\begin{prop}
		Dadas $f_i : U \to \mathbb{R}$ convexas y $a_i \geq 0$ para todo $i$ entonces $f = \sum_{i} a_i f_i$ es convexa.
 	\end{prop}
 	
 	\begin{proof}
 		Hacer la cuenta...
 	\end{proof}
 	
 	\begin{prop}
 		Sea $f : U \subset \mathbb{R}^{n} \to \mathbb{R}$ convexa, entonces el conjunto $\Gamma_C = \{x \in U : f(x) \leq C\}$ es convexo para cualquier $C$. 
 	\end{prop}
	
	\begin{proof}
		Queremos ver que $\Gamma_C$ es convexo, tomemos entonces $x, y \in \Gamma_C$, $\lambda \in [0, 1]$ y veamos que $z = \lambda x + (1- \lambda)y \in \Gamma_C$ pues $f(z) \leq \lambda f(x) + (1- \lambda) f(y) \leq C.$ 
	\end{proof}
	
	\begin{ej}[La norma es convexa]
		Dado $V$ un espacio vectorial si $\|\cdot\| : V \to \mathbb{R}$ es una norma, entonces es convexa. ¿Por qué? Por la desigualdad triangular. 
		Este resultado nos permite ver que la bola en cualquier norma es convexa.
	\end{ej}
	
	\subsubsection{Funciones convexas diferenciables.}
	
	\begin{prop}
		Dada $f : \Omega \to \mathbb{R}$ con $f \in C^{1}(\Omega)$ son equivalentes:
		\begin{enumerate}
			\item $f$ es convexa.
			\item Dados $x, y \in \Omega$ se tiene que $f(x) \geq f(y) + \nabla f(y)^{t} (x - y).$
		\end{enumerate}
	\end{prop}
 	
 	\begin{obs}
 		Notemos que los roles de $x$ e $y$ en la segunda condición son intercambiables.
 	\end{obs}
 	
 	\begin{proof}
 		Veamos primero que $1.$ implica $2.$ pues dados $x, y \in \Omega$, dado $\lambda \in [0, 1]$ sabemos que
 		$$
 		f(\lambda x + (1- \lambda) y) \leq \lambda f(x) + (1 - \lambda) f(y) 
 		$$ 
 		reescribiendo tenemos que esto es
 		$$ 
 		\frac{f(y + \lambda(x - y)) - f(y)}{\lambda} \leq f(x) - f(y) 
 		$$
 		consideremos $\phi : [0, 1] \to \mathbb{R}$ tal que $\phi(\lambda) = f(y + \lambda(x - y))$. De esta manera tenemos que la expresión anterior se puede escribir como
 		$$
 		\frac{\phi(\lambda) - \phi(0)}{\lambda} \leq f(x) - f(y) 
 		$$
 		tomando límite con $\lambda \to 0$,
 		$$
 		\nabla f(y)^{t} (x - y) = \phi'(0) \leq f(x) - f(y).
 		$$
 		Nos queda probar ahora que $2.$ implica $1.$, tomando $z = \lambda x + (1 - \lambda)y$ y utilizando la observación tenemos que
 		\[
 		\begin{aligned}
 		f(x) \geq f(z) + \nabla f(z)^{t} (x - z) \\
 		f(y) \geq f(z) + \nabla f(z)^{t} (y - z)	
 		\end{aligned}
 		\]
 		multiplicando por $\lambda$ la primer desigualdad, por $1-\lambda$ la segunda y sumando tenemos
 		\[
 		\begin{aligned}
 			\lambda f(x) + (1- \lambda)f(y) &\geq f(z) + \lambda \nabla f(z)^{t} (x - z) + (1-\lambda) \nabla f(z)^{t} (y - z) \\
 			&= f(z) + \nabla f(z)^{t} (\lambda x - \lambda z + (1 - \lambda) y - (1 - \lambda) z) \\
 			&= f(z) + \nabla f(z)^{t} (\lambda x + (1 - \lambda) y - z) \\
 			&= f(z)
 		\end{aligned}
 		\] 
 		que es lo que queríamos ver.
 	\end{proof}
 	
 	\begin{prop}
 		Sea $f \in C^{1}(K)$ convexa, entonces 
 		$$
 		\nabla f(x_1)^{t}(x_2 - x_1) \leq \nabla f(x_2)^{t} (x_2 - x_1)
 		$$
 		para todo $x_1, x_2 \in K$.
 	\end{prop}
 	
 	\begin{proof}
 		Sabemos, o deberíamos saber, que dados $x_1, x_2 \in K$
 		\[
 		\begin{aligned}
 			f(x_2) &\geq f(x_1) + \nabla f(x_1)^{t} (x_2 - x_1) \text{ y, } \\
 			f(x_1) &\geq f(x_2) + \nabla f(x_2)^{t} (x_1 - x_2)
 		\end{aligned}
 		\]
 		de esta manera, 
 		\[
 		\begin{aligned}
 			f(x_2) - f(x_1) &\geq \nabla f(x_1)^{t} (x_2 - x_1) \text{ y, } \\
 			f(x_2) - f(x_1) &\leq \nabla f(x_2)^{t} (x_1 - x_2)
 		\end{aligned}
 		\]
 		uniendo ambas desigualdades obtenemos el resultado.
 	\end{proof}
 	
 	\begin{prop}
 		Si $K \subset \mathbb{R}^{n}$ es convexo y $f \in C^{2}(K)$. Son equivalentes:
 		\begin{enumerate}
 			\item $f$ es convexa.
 			\item $\forall x_1, x_2 \in K$ tenemos que $d^{t}\nabla^{2}f(x_1)d \geq 0$ donde $d = x_2 - x_1.$
 		\end{enumerate}
 	\end{prop}
 	
 	\begin{proof}
 		$2. \implies 1.$  Por Taylor sabemos que
 		$$
 		f(x_2) = f(x_1) + \frac{1}{2} \nabla f(x_1)^{t}d +d^{t}\nabla^2 f(\xi)d 
 		$$
 		con $\xi$ una combinación convexa de $x_1, x_2$. La idea ahora es notar que $d = t(x_2 - \xi),$ con $t \geq 0.$ De esta manera,
 		$$
 		d^{t}\nabla^2 f(\xi)d = t^{2} (x_2- \xi)\nabla^2 f(\xi)(x_2 - \xi) \geq 0.
 		$$ 
 		Luego, $f$ es convexa. \\
 		
 		$2. \implies 1.$  Fijemos $x_1$ en $K$ y definamos $g(x) = f(x) - \nabla f(x_1)^t x$ y, notemos que $g$ tiene un mínimo en $x_1$. Además,
 		\begin{enumerate}
 			\item $\nabla g(x_1) = 0$.
 			\item $\nabla^{2} g(x_1) = \nabla^{2} f(x_1).$ 
 		\end{enumerate}
 		Luego, 
 		$$
 		0 \leq g(x) - g(x_1) = \frac{1}{2} (x - x_1) ^{t}\nabla^2 f(x) (x - x_1) + o(\|x - x_1\|),
 		$$
 		tomando $x = t(x_2 - x_1) + x_1$, si $t \in (0, 1)$ entonces $x \in K$ y,
 		$$
 		0 \leq g(x) - g(x_1) = \frac{t^2}{2} (x_2 - x_1) ^{t}\nabla^2 f(x) (x_2 - x_1) + \frac{o(t^2)}{t^2},
 		$$
 		diviendo por $t^2$ y tomando límite con $t \to 0$ tenemos el resultado.
 	\end{proof}
 	
 	\section{Algoritmos y convergencia.}
 	
 	El objetivo de este capítulo -- y de esta materia -- es minimizar funciones. Considero relevante comenzar con un ejemplo sencillo que muestre qué tipo de métodos aparecen minimizando funciones. Supongamos entonces que queremos minimizar $q(x) = \frac{1}{2} x^t G x - b^t x + c$. Podemos suponer que $G$ es simétrica, sino basta cambiarla por $\frac{1}{2} (G + G^t).$ ¿Qué sabemos de esta función?
	\begin{itemize}
		\item $\nabla q(x) = Gx - b,$
		\item $\nabla^2 q(x) = G.$ 
	\end{itemize}
	Esto nos dice que, en caso de existir, un punto crítico $x^*$ satisface que $Gx^* = b$. Cuando
	$b\in$ subespacio generado por las columnas de $G$ esto tiene solución y, es única cuando $G$ es inversible. \\
	\indent Supongamos ahora que $G > 0$, una idea para minimizar es considerar la dirección de máximo descenso en un punto $x_k$ que viene dada por 
	$$d_k = - \nabla q(x_k) = - g_k,$$
	y dar un paso de cierta longitud $\alpha_k$ en esa dirección, 
	$$ 
	x_{k+1} = x_{k} + \alpha_{k} d_k
	$$
	e iterar. A esto lo llamamos \emph{algoritmo de descenso.} En este caso, podemos aprovechar que sabemos que es una función cuadrática, veamos. Si definimos $h(\alpha) = q(x_k + \alpha d_k)$ sabemos que es una parábola que viene dada por,
	$$
	h(\alpha) = \frac{1}{2} \alpha^2 d_k^t G d_k + \alpha d_k^t G x_k + \frac{1}{2} x_k^t G x_k - b^t x_k - \alpha b^t x_k + c 
   	$$
   	Esta tiene un mínimo en $\alpha^*$ si $h'(\alpha^*) = 0$, esto es 
   	$$
   	\alpha^* d_k^t G_k d_k + d_k^t G x_k - b^t d_k = \alpha^* d_k^t G_k d_k + d_k^t(G x_k - b) = \alpha^* d_k^t G_k d_k + d_k^tg_k,
    $$	
    Tomando $\alpha_k = \alpha^*$ podemos considerar,
	\[
	\left\{
	\begin{aligned}
		x_{k+1} &= x_k - \alpha_k g_k \\
		\alpha_k &= \frac{- g_k^\top g_k}{g_k^\top G \, g_k}
	\end{aligned}
	\right.
	\]
	tenemos así nuestro algoritmo para minimizar. ¿Es útil? No, pero aporta valor teórico para comprender los métodos de descenso.
	
	\subsection{Convergencia global de algoritmos de descenso.}
	
	A lo largo del curso, será de interés estudiar diversos algoritmos y, en particular, su convergencia. En general, estudiaremos algoritmos iterativos y daremos un marco teórico que permita estudiarlos sin preocuparnos por implementaciones particulares. Para este capítulo, vamos a basarnos en \cite{luenberger2008nonlinear}.
	
	\subsection{Algoritmos iterativos.}
	
	Vamos a pensar a los algoritmos como mapeos. Dado un punt $x$ en un cierto espacio $X$ 
	la salida de un algoritmo aplicado a $x$ es un punto. Aplicando el algoritmo a este nuevo 
	punto iterativamente concstruimos una sucesión de puntos. Esta definición preeliminar está bien, pero
	queremos más flexibilidad en nuestra definción. Esto da lugar a la siguiente definición.
	
	\begin{definicion}
		Un algoritmo $A$ es un mapa definido en un espacio $X$ que asigna a cada punto $x \in X$
		un subconjunto de $X.$
	\end{definicion}
	
	El termino espacio puede ser interpretado vagamente, usualmente este será un espacio vectorial
	pero puede ser solo un subconjunto o inclusive ser un espacio métrico.
	
	¿No es un poco ambigua esta definición? En la práctica un algoritmo es un mapa de punto a punto, cuál
	es el sentido de definirlo así. La utilidad de esta definición es la de poder analizar, en un solo 
	paso una familia infinita de alogritmos similares.
	
	\subsection{Descenso.}
	
	Queremos describir que es un algortimo de descenso, para eso necesitamos algunas definiciones previas.
	
	\begin{definicion}
		Sea $\Gamma \subset X$ un conjunto solución y sea $A$ un algoritmo en $X$. Una función continua de $X \to \mathbb{R}$ es una \emph{función de descenso} para $(\Gamma, A)$ si:
		
		\begin{enumerate}
			\item si $x \notin \Gamma$ e $y \in A(x)$, entonces $Z(y) < Z(x),$
			\item si $x \in \Gamma$ e $y \in A(x)$, entonces $Z(y) \leq Z(x).$
		\end{enumerate}
		
	\end{definicion}
	
	\subsection{Mapas cerrados.}
	
	Necesitamos, en nuestro estudio de convergencia de algoritmos, una noción
	de compacidad.
	
	\begin{definicion}
		Un mapa de punto a conjunto $A$ de $X$ en $Y$ se dice cerrado en $x \in X$ si
		\begin{enumerate}
			\item $x_k \to x$ con $x_k \in X$,
			\item $y_k \to y$ con $y_k \in Ax_k$ 
		\end{enumerate}
		implican que $y \in Ax$.
	\end{definicion}

	Los algoritmos que estudiemos van a ser, en general, composición de algoritmos. ¿Qué significa 
	componer con nuestra definición? Lo que uno se imagina.
	
	\begin{definicion}
		Sean $A : X \to Y$, $B: Y \to Z$ mapas de punto a conjunto. Definimos el mapa de punto a conjunto
		composición como $BA : X \to Z$ como
		$$
		BA(x) = \bigcup_{y \in A(x)} B(y)
		$$
	\end{definicion}
	
	
	\begin{prop}
		Sean $A: X \to Y$, $B : Y \to Z$ mapas de punto a conjunto. Supongamos que $A$ es cerrado en $x$ 
		y que $B$ es cerrado en $A(x)$. Supongamos además que si $x_k \to x$ e $y_k \in A(x_k)$ entonces existen $y$, una subsucesión $\{ y_{k_j}\}$ tal que $y_{k_j} \to y$. Luego el mapa $BA$ es cerrado en $x$.  
	\end{prop}
	
	\begin{proof}
		Sea $x_k \to x$ y $z_k \to z$ con $z_k \in BA(x_k)$. Queremos ver que $z \in BA(x)$.
		Sea $y_k \in A(x_k)$ tal que $z_k \in BA(x_k)$, por hipótesis tenemos $y$, $\{y_{k_j}\}$
		tal que $y_{k_j} \to y$. Como $A$ es cerrado en $x$ entonces $y \in A(x)$. De la misma manera, 
		como $B$ es cerrado en $y$ pues lo es en $A(x)$ entonces $z \in B(y) \subset BA(x).$
	\end{proof}
	
	Corollary 2. Let A : X → Y be a point-to-point mapping and B : Y → Z a point-to
	set mapping. If A is continuous at x and B is closed at A(x), then the composite mapping
	C=BAisclosed at x.
	
	
	\begin{coro}
		Sean $A: X \to Y$, $B : Y \to Z$ mapas de punto a conjunto. Si $A$ es cerrado en $x$, $B$ es
		cerrado en $A(x)$ e $Y$ es compacto, luego la composición es compacta en $x.$
	\end{coro}
	
	\begin{coro}
			Sean $A: X \to Y$ un mapa de punto a punto, $B : Y \to Z$ un mapa de punto a conjunto. Si
			$A$ es continua en $x$ y $B$ es cerrado en $A(x),$ luego la composición es cerrada en $x.$  
	\end{coro}

	\subsection{Teorema de convergencia global.}
	
	\begin{teo}[Convergencia global.]
		Sea $A$ un algoritmos en $X$, y supongamos que, dado $x_0$ la sucesión
		$\{x_k\}_{k \geq 0}$ satisface que $x_{k+1} \in A(x_k)$. Sea $\Gamma \subset X$ un conjunto solución
		y supongamos que
		\begin{enumerate}
			\item todos los $x_k$ están contenidos en un compacto $S \subset X,$
			\item existe una función continua $Z$ en $X$ tal que
			\begin{itemize}
				\item si $x \notin \Gamma$, luego $Z(y) < Z(x)$ para todo $y \in A(x),$
				\item si $x \in \Gamma,$ entonces $Z(y) \leq Z(x)$ para todo $y \in A(x)$
			\end{itemize} 
			\item el mapa $A$ es cerrado para los puntos en $X \setminus \Gamma.$ 
		\end{enumerate}
		Luego, el limite de cualquier subsucesión convergente $x_k$ es una solución.
	\end{teo}
	
	\begin{proof}
		Sea $\{x_k\}_{k \in \mathcal{K}}$ una subsucesión que converge a un límite $x$. Como $Z$ es continua,
		tenemos que para $k \in \mathcal{K}$, $Z(x_k) \to Z(x).$ Como $Z$ es monótona decreciente, tenemos que 
		$Z(x_k) - Z(x) \geq 0$ para todo $k.$ Dado $\varepsilon > 0$, existe $K \in \mathcal{K}$ tal que $Z(x_k) - Z(x) < \varepsilon$ para todo $k \geq K, k \in \mathcal{K}.$ Luego, para todo $k > K$
		$$
		Z(x_k) - Z(x) =  Z(x_k) - Z(x_K) + Z(x_K) - Z(x) < \varepsilon
		$$
		así, $Z(x_k) \to Z(x)$ para todo $k$. Resta ver que $x$ es una solución. Supongamos que no lo es,
		y consideremos $\{x_{k+1}\}_{k \in \mathcal{K}}$ como está contenida en un compacto, entonces existe $\hat{\mathcal{K}} \subset \mathcal{K}$ tal que $\{x_{k+1}\}_{k \in \hat{\mathcal{K}}}$ converge a un límite $\hat{x}.$ Tenemos así, $x_{k} \to x$ con $k \in \hat{\mathcal{K}}$ y $x_{k+1} \in A(x_k)$ con $x_{k+1} \to \hat{x},$ con $k \in \hat{\mathcal{K}}.$ Cómo $A$ es cerrado en $x$ tenemos que $\hat{x} \in A(x)$. Pero, por lo que vimos antes, $Z(x) = Z(\hat{x}),$ lo cual contradice el hecho de que $Z$ sea una función de descenso.
		
	\end{proof}
	
	\subsection{Velocidad de convergencia.}
	
	En esta sección vamos a introducir conceptos relativos a la velocidad de convergencia que nos van a permitir predecir con cierta confianza la efectividad de una amplia clase de algoritmos.
	
	\subsection{Orden de convergencia}
		
	\begin{definicion}
		Sea $\{r_k\}_k$ una sucesiín que converge a $r^{*}.$ El orden de convergencia de esta sucesión se define como el supremo de los $p \geq 0$ que satisfacen
		$$
		\limsup_{k\to \infty} \frac{|r_{k+1} - r^{*}|}{|r_k - r^{*}|^{p}} 
		$$ 
	\end{definicion}
		
	Podemos decir que el orden de convergencia es una medida de que tan buena es la peor parte de la cola de la serie. Valores grandes de $p$ nos hablan de una convergencia más rapida. Es decir, si la sucesión tiene orden $p$ tenemos que asintoticamente
	$$
	|r_{k+1} - r^*| = \beta|r_k - r^*|^{p} 
	$$
	donde $\beta$ es el límite del cociente que teniamos arriba.
	
	\subsection{Convergencia lineal}
	
	\begin{definicion}
		Si la sucesión $\{r_k\}_k$ converge a $r^*$ de manera que
		$$
		\lim_{k\to \infty} \frac{|r_{k+1} - r^{*}|}{|r_k - r^{*}|} = \beta 
		$$
		decimos que la sucesión converge linealmente con orden $\beta$.
	\end{definicion}
	
	Además, si $\beta$ es 0, decimos que la convergencia es \emph{superlineal}. Notemos que si $p > 1$ entonces la convergencia es superlineal. Este marco nos permite comparar algoritmos mirando el ratio de convergencia. Existen otros tipos de convergencia, como la aritmética pero no vamos a adentrarnos en estos.
	
	\section{Algoritmos básicos de descenso.}
	
	En este capítulo vamos a describir algunas técnicas básicas para resolver iterativamente problemas de minimización sin restricciones. Estas técnicas son importantes por su simpleza y por sentar bases para proximos métodos.
	
	Los algoritmos de descenso que vamos a estudiar comparten estructura. Arrancamos en un punto, determinamos de acuerdo a alguna regla una dirección 
	de movimiento y luego nos movemos en esa dirección hacia algún mínimo relativo en esa dirección. A partir de ese nuevo punto, repetimos el proceso.
	La principal diferencia entre algoritmos es esta regla de elección de direcciones. El proceso de determinar un mínimo en una recta dada (una variable) se llama búsqueda lineal y usualmente se realiza de alguna manera inteligente, buscando quizás una solución inexacta.
	
	\subsection{Algortimos de busqueda lineal.}
	
	Estos se clasifican según el orden de información que requieren de nuestra función a minimizar.
	
	\subsubsection{Orden 0: Fibonacci y fitteo de curvas.}
	
	Un método muy utilizado para resolver el problema de búsqueda lineal es el \emph{método de búsqueda de Fibonacci}. Aunque posee cierta elegancia teórica, existen otros procedimientos que, en la mayoría de los casos, resultan superiores.
	
	El método determina el mínimo de una función $f$ sobre un intervalo cerrado $[c_1, c_2]$. La única propiedad que se asume sobre $f$ es que es \emph{unimodal}, es decir, posee un único mínimo relativo.
	
	
	La idea central es reducir progresivamente el intervalo de búsqueda evaluando $f$ en puntos estratégicamente elegidos. En cada iteración se seleccionan dos puntos interiores $x_a$ y $x_b$ con $c_1 < x_a < x_b < c_2$, y se comparan sus valores:
	
	\begin{itemize}
		\item Si $f(x_a) < f(x_b)$, el mínimo se encuentra en $[c_1, x_b]$, descartando el subintervalo derecho.
		\item Si $f(x_a) > f(x_b)$, el mínimo se encuentra en $[x_a, c_2]$, descartando el subintervalo izquierdo.
	\end{itemize}
	
	
	Sea $L_k$ la longitud del intervalo activo en la iteración $k$, y sean $F_1, F_2, \ldots, F_N$ los números de Fibonacci definidos por:
	\[
	F_1 = F_2 = 1, \qquad F_k = F_{k-1} + F_{k-2}, \quad k \geq 3.
	\]
	
	Supóngase que se dispone de un total de $N$ evaluaciones de $f$. En la iteración $k$ (con $k = 1, 2, \ldots, N-1$), los dos puntos de evaluación sobre el intervalo activo $[a_k, b_k]$ de longitud $L_k = b_k - a_k$ se ubican en:
	\[
	x_a^{(k)} = a_k + \frac{F_{N-k-1}}{F_{N-k+1}}\, L_k, \qquad
	x_b^{(k)} = a_k + \frac{F_{N-k}}{F_{N-k+1}}\, L_k.
	\]
	
	La propiedad clave de esta elección es que uno de los dos puntos evaluados en la iteración $k$ coincide con un punto ya evaluado en la iteración $k-1$, de modo que solo se requiere una nueva evaluación por iteración (salvo en la primera). Esto hace que con $N$ evaluaciones en total se obtenga una reducción del intervalo de incertidumbre en la proporción:
	\[
	\frac{L_N}{L_1} = \frac{1}{F_N}.
	\]
	
	Este cociente es el mejor posible para cualquier método que evalúe $f$ en $N$ puntos sin conocimiento previo de su forma, lo que confiere al método de Fibonacci su carácter óptimo en este sentido.
	
	El esquema más útil en la búsqueda lineal consiste en ajustar una parábola a través de tres puntos dados. Esto tiene la ventaja de no requerir información sobre derivadas. Dados $x_1, x_2, x_3$ y los valores correspondientes $f(x_1) = f_1$, $f(x_2) = f_2$, $f(x_3) = f_3$, se construye el polinomio cuadrático que pasa por estos puntos:
	
	\begin{equation}
		q(x) = \sum_{i=1}^{3} f_i \prod_{j \neq i} \frac{x - x_j}{x_i - x_j},
		\label{eq:8.7}
	\end{equation}
	
	y se determina un nuevo punto $x_4$ como aquel donde se anula la derivada de $q$:
	
	\begin{equation}
		x_4 = \frac{1}{2} \cdot \frac{b_{23}f_1 + b_{31}f_2 + b_{12}f_3}{a_{23}f_1 + a_{31}f_2 + a_{12}f_3},
		\label{eq:8.8}
	\end{equation}
	
	donde $a_{ij} = x_i - x_j$ y $b_{ij} = x_i^2 - x_j^2$.
	
	Se definen los errores $\varepsilon_i = x^* - x_i$, para $i = 1, 2, 3, 4$, donde $x^*$ es el mínimo de $f$. La expresión para $\varepsilon_4$ debe ser un polinomio en $\varepsilon_1, \varepsilon_2, \varepsilon_3$ que satisfaga las siguientes propiedades:
	
	\begin{itemize}
		\item Es de segundo orden, pues proviene de un ajuste cuadrático.
		\item Se anula si cualquiera de los dos errores $\varepsilon_1, \varepsilon_2, \varepsilon_3$ es cero.
		\item Es simétrico, ya que el orden de los puntos es relevante.
	\end{itemize}
	
	De estas propiedades se concluye que, cerca del mínimo $x^*$, los errores se relacionan aproximadamente por:
	
	\begin{equation}
		\varepsilon_4 = M(\varepsilon_1\varepsilon_2 + \varepsilon_2\varepsilon_3 + \varepsilon_1\varepsilon_3),
		\label{eq:8.9}
	\end{equation}
	
	donde $M$ depende de los valores de las derivadas segunda y tercera de $f$ en $x^*$.
	
	Suponiendo que $\varepsilon_k \to 0$ con un orden mayor que la unidad, para $k$ grande el error se rige aproximadamente por:
	\[
	\varepsilon_{k+2} = M\,\varepsilon_k\,\varepsilon_{k-1}.
	\]
	
	Definiendo $y_k = \log(M\varepsilon_k)$, esta relación se transforma en:
	\[
	y_{k+2} = y_k + y_{k-1},
	\]
	
	cuya ecuación característica es:
	\begin{equation}
		\lambda^3 - \lambda - 1 = 0.
	\end{equation}
	
	La raíz más grande de esta ecuación es $\lambda \approx 1.3$, lo cual determina la tasa de crecimiento de $y_k$ y constituye el orden de convergencia del método de ajuste cuadrático.
	
	No vamos a ahondar en los métodos de primer orden, por lo que resulta conveniente pasar directamente a los métodos de orden 2.
	
	\subsubsection{Orden 2: Método de Newton.}
	
	Supóngase que se desea minimizar una función $f$ de una sola variable $x$, y que en un punto $x_k$ es posible evaluar los tres valores $f(x_k)$, $f'(x_k)$ y $f''(x_k)$. Con esta información se construye una función cuadrática $q$ que coincide con $f$ en $x_k$ hasta las derivadas de segundo orden:
	
	\begin{equation}
		q(x) = f(x_k) + f'(x_k)(x - x_k) + \frac{1}{2}\,f''(x_k)(x - x_k)^2.
		\label{eq:8.20}
	\end{equation}
	
	El estimado $x_{k+1}$ del punto mínimo de $f$ se obtiene hallando el punto donde se anula la derivada de $q$. Imponiendo la condición
	
	\begin{equation}
		0 = q'(x_{k+1}) = f'(x_k) + f''(x_k)(x_{k+1} - x_k),
	\end{equation}
	
	se obtiene la iteración de Newton
	
	\begin{equation}
		x_{k+1} = x_k - \frac{f'(x_k)}{f''(x_k)}.
		\label{eq:newton}
	\end{equation}
	
	Este método requiere el cálculo de la primera y segunda derivada de $f$ en cada iteración, pero a cambio ofrece convergencia cuadrática cerca del mínimo, lo que lo hace significativamente más rápido que los métodos de orden cero cuando la evaluación de derivadas es factible.
	
	Para aplicar el método de Newton a la minimización de $f$, se observa que los puntos críticos satisfacen $g(x) = 0$ donde $g(x) = f'(x)$. La iteración de Newton es
	\[
	x_{k+1} = x_k - \frac{g(x_k)}{g'(x_k)} = x_k - \frac{f'(x_k)}{f''(x_k)},
	\]
	que coincide con \eqref{eq:newton}. La siguiente proposición establece la convergencia del método.
	
	\begin{prop}
		Sea $g = f'$ una función con segunda derivada continua, y sea $x^*$ tal que $g(x^*) = 0$ y $g'(x^*) \neq 0$. Entonces, si $x_0$ es suficientemente cercano a $x^*$, la sucesión $\{x_k\}_{k=0}^{\infty}$ generada por el método de Newton converge a $x^*$ con un orden de convergencia de al menos dos.
	\end{prop}
	
	\begin{proof}
		En una región cercana a $x^*$ existen constantes $k_1$ y $k_2$ tales que:
		\[
		|g''(\xi)| < k_1 \qquad \text{y} \qquad |g'(\xi)| > k_2 > 0
		\]
		para todo $\xi$ en dicha región. Dado que $g(x^*) = 0$, se puede escribir:
		\[
		x_{k+1} - x^* = x_k - x^* - \frac{g(x_k) - g(x^*)}{g'(x_k)}
		= -\frac{g(x_k) - g(x^*) + g'(x_k)(x^* - x_k)}{g'(x_k)}.
		\]
		
		El término en el numerador es, por el teorema de Taylor, de orden superior al primero. Aplicando el resto de la expansión de Taylor alrededor de $x_k$, se obtiene:
		\[
		x_{k+1} - x^* = \frac{1}{2}\,\frac{g''(\xi)}{g'(x_k)}\,(x_k - x^*)^2,
		\]
		para algún $\xi$ entre $x^*$ y $x_k$. Por lo tanto, en la región cercana a $x^*$:
		\begin{equation}
			|x_{k+1} - x^*| \leq \frac{k_1}{2k_2}\,|x_k - x^*|^2.
			\label{eq:newton_conv}
		\end{equation}
		
		Si $|x_k - x^*|\cdot k_1/2k_2 < 1$, entonces $|x_{k+1} - x^*| < |x_k - x^*|$, y se concluye que, partiendo suficientemente cerca de la solución, el método converge a $x^*$ con un orden de convergencia de al menos dos.
	\end{proof}
	
	\subsection{Clausura de algoritmos de busqueda lineal.}
	
	\begin{definicion}
		El mapeo $S : \R^{2n} \to \R^n$ se define como:
		\begin{equation}
			S(x, d) = \left\{ y : y = x + \alpha d \ \text{para algún} \ \alpha \geq 0, \
			f(y) = \min_{0 \leq \alpha < \infty} f(x + \alpha d) \right\}.
			\label{eq:8.23}
		\end{equation}
	\end{definicion}
	
	En algunos casos puede haber múltiples vectores $y$ que alcancen el mínimo, por lo que $S$ es un mapeo multivaluado. Es necesario verificar que $S$ es cerrado.
	
	\begin{teo}
		Sea $f$ continua en $\R^n$. Entonces el mapeo definido por \eqref{eq:8.23} es cerrado en $(x, d)$ si $d \neq 0$.
	\end{teo}
	
	\begin{proof}
		Sean $\{x_k\}$ y $\{d_k\}$ sucesiones tales que $x_k \to x$ y $d_k \to d \neq 0$. Supongamos además que $y_k \in S(x_k, d_k)$ e $y_k \to y$. Queremos ver que $y \in S(x, d)$.
		
		Para cada $k$, existe $\alpha_k \geq 0$ tal que $y_k = x_k + \alpha_k d_k$, de donde:
		\[
		\alpha_k = \frac{|y_k - x_k|}{|d_k|}.
		\]
		Tomando el límite del lado derecho, se obtiene:
		\[
		\alpha_k \to \alpha \equiv \frac{|y - x|}{|d|},
		\]
		y en consecuencia $y = x + \alpha d$.
		
		Resta demostrar que $y \in S(x, d)$. Para cada $k$ y cada $\alpha \geq 0$:
		\[
		f(y_k) \leq f(x_k + \alpha d_k).
		\]
		Tomando el límite $k \to \infty$ y usando la continuidad de $f$:
		\[
		f(y) \leq f(x + \alpha d).
		\]
		Como esto vale para todo $\alpha \geq 0$, se concluye:
		\[
		f(y) \leq \min_{0 \leq \alpha < \infty} f(x + \alpha d),
		\]
		y por lo tanto $y \in S(x, d)$.
	\end{proof}
	
	La condición $d \neq 0$ es natural tanto teórica como prácticamente. Desde el punto de vista práctico, implica que al construir algoritmos la elección $d = 0$ debe ocurrir únicamente en el conjunto solución, pues si $d = 0$ no se realiza ninguna búsqueda. Teóricamente, el mapeo $S$ puede no ser cerrado en $d = 0$.
	
	\subsection{Busqueda inexacta.}
	
	En la práctica sabemos que no es posible encontrar mínimos fácilmente en cada busqueda lineal, por lo que introducimos el concepto de busqueda inexacta. En esta, terminamos de iterar antes de obtener el mínimo basandonos en reglas que nos aseguren ciertas propiedades sobre el paso.
	
	\subsubsection{Armijo.}
	
	Un criterio práctico y popular para terminar una búsqueda lineal es la regla de Armijo. La idea esencial es garantizar que el paso $\alpha$ seleccionado no sea ni demasiado grande ni demasiado pequeño. Se define
	\[
	\varphi(\alpha) = f(x_k + \alpha d_k).
	\]
	
	La regla se implementa considerando recta
	\[
	\varphi(0) + \varepsilon\,\varphi'(0)\,\alpha, \qquad \varepsilon \in (0, 1) \ \text{fijo}.
	\]
	Un valor de $\alpha$ se considera no demasiado grande si el valor de la función cae por debajo de dicha recta, es decir, si
	
	\begin{equation}
		\varphi(\alpha) \leq \varphi(0) + \varepsilon\,\varphi'(0)\,\alpha.
		\label{eq:armijo}
	\end{equation}
	
	Para garantizar que $\alpha$ no sea demasiado pequeño, se selecciona un valor $\eta > 1$ y se considera que $\alpha$ es no demasiado pequeño si:
	\[
	\varphi(\eta\alpha) > \varphi(0) + \varepsilon\,\varphi'(0)\,\eta\alpha.
	\]
	
	En la práctica, la regla de Armijo se utiliza para definir una búsqueda lineal simplificada sin ajuste de curvas. El procedimiento es el siguiente, con valores típicos $\eta = 2$ o $\eta = 10$ y $\varepsilon = 0.2$:
	
	\begin{enumerate}
		\item Se parte de un $\alpha$ arbitrario.
		\item Si $\alpha$ satisface \eqref{eq:armijo}, se lo incrementa repetidamente por $\eta$ hasta que \eqref{eq:armijo} deje de cumplirse; luego se selecciona el penúltimo valor de $\alpha$.
		\item Si $\alpha$ no satisface \eqref{eq:armijo}, se lo divide repetidamente por $\eta$ hasta que sí lo satisfaga.
	\end{enumerate}
	
	\subsubsection{Condiciones de Wolfe}
	
	Las condiciones de Wolfe combinan la regla de Armijo con una condición adicional sobre la derivada, logrando un control más preciso sobre el tamaño del paso. Dado $\varepsilon \in (0, 1)$ y $\sigma \in (\varepsilon, 1)$, las condiciones son
	
	\begin{equation}
		\varphi(\alpha) \leq \varphi(0) + \varepsilon\,\varphi'(0)\,\alpha,
		\label{eq:wolfe1}
	\end{equation}
	\begin{equation}
		\varphi'(\alpha) \geq \sigma\,\varphi'(0).
		\label{eq:wolfe2}
	\end{equation}
	
	La condición \eqref{eq:wolfe2} exige que la pendiente de $\varphi$ en $\alpha$ no sea demasiado negativa, lo que impide que el paso sea demasiado corto.
	
	
	\subsubsection{Condiciones de Goldstein}
	
	Una alternativa a las condiciones de Wolfe es la regla de Goldstein, que controla simultáneamente que el paso $\alpha$ no sea ni demasiado grande ni demasiado pequeño mediante dos desigualdades simétricas. Dado $\varepsilon \in (0, \tfrac{1}{2})$, las condiciones de Goldstein son
	
	\begin{equation}
		\varphi(0) + (1-\varepsilon)\,\varphi'(0)\,\alpha 
		\leq \varphi(\alpha) \leq 
		\varphi(0) + \varepsilon\,\varphi'(0)\,\alpha.
		\label{eq:goldstein}
	\end{equation}
	
	La desigualdad derecha coincide con la condición de Armijo \eqref{eq:armijo} y garantiza un descenso suficiente, evitando que $\alpha$ sea demasiado grande. La desigualdad izquierda, en cambio, impone una cota inferior sobre el decrecimiento de $\varphi$, evitando que $\alpha$ sea demasiado pequeño.
